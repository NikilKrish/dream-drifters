function toggleSub(id){
  const sub=document.getElementById('sub-'+id);
  const btn=sub.previousElementSibling;
  const isOpen=sub.classList.contains('open');
  document.querySelectorAll('.pkg-sub').forEach(s=>s.classList.remove('open'));
  document.querySelectorAll('.pkg-option').forEach(b=>b.classList.remove('active'));
  if(!isOpen){sub.classList.add('open');btn.classList.add('active');}
}

  el.classList.add('selected');
  document.getElementById('packageInput').value=val;
  const d=document.getElementById('selectedDisplay');
  d.textContent='Selected: '+val;
  d.classList.add('show');
}
